using System;
using System.Reflection;
using UnityEngine;

// Stock coloured Variant UI controls the existing PWR part. The maintained
// finish module remains independent. Reflection keeps Kerbalism optional.
public sealed class ModuleRWPowerVariant : PartModule
{
    [KSPField] public double batteryCapacity = 3500;
    [KSPField] public double reactorCapacity = 3062.5;
    [KSPField] public double reactorScale = 1.1;
    private bool? applied;
    private ModulePartVariants variants;
    private PartModule controller;
    public override void OnStart(StartState state)
    {
        base.OnStart(state);
        variants = part.FindModuleImplementing<ModulePartVariants>();
        foreach (PartModule m in part.Modules)
        {
            var field = m.GetType().GetField("resource");
            if (m.GetType().Name == "ProcessController" && field != null && (string)field.GetValue(m) == "_RWCompactReactor") controller = m;
        }
        Reconcile();
    }
    public void Update()
    {
        if (HighLogic.LoadedSceneIsEditor || HighLogic.LoadedSceneIsFlight) Reconcile();
    }
    private void Reconcile()
    {
        if (part == null || controller == null || variants == null || variants.SelectedVariant == null) return;
        bool nuclear = variants.SelectedVariant.Name == "Nuclear";
        bool changed = !applied.HasValue || applied.Value != nuclear;
        bool resourcesChanged = changed;
        if (changed)
        {
            // Editor exchanges hardware. Flight restores existing inventory;
            // missing fuel is empty, never regenerated by starting/loading.
            Resize("ElectricCharge", nuclear ? reactorCapacity : batteryCapacity, 0);
            if (nuclear)
            {
                Resize("EnrichedUranium", 1.5 * reactorScale, HighLogic.LoadedSceneIsEditor ? 1.5 * reactorScale : 0);
                Resize("DepletedFuel", 1.5 * reactorScale, 0);
                if (HighLogic.LoadedSceneIsEditor) SetRunning(false);
            }
            else if (HighLogic.LoadedSceneIsEditor)
            {
                part.Resources.Remove("EnrichedUranium");
                part.Resources.Remove("DepletedFuel");
            }
            applied = nuclear;
            Debug.Log("[RoosterWorks] PWR variant " + (nuclear ? "Nuclear" : "Battery") + ": " + (nuclear ? reactorCapacity : batteryCapacity) + " EC.");
        }
        // Other editor listeners can restore prefab capacities after a variant
        // event. Check actual storage, not only our remembered selection.
        double targetCapacity = nuclear ? reactorCapacity : batteryCapacity;
        PartResource charge = part.Resources["ElectricCharge"];
        if (charge == null || Math.Abs(charge.maxAmount - targetCapacity) > 1e-7)
        {
            Resize("ElectricCharge", targetCapacity, 0);
            resourcesChanged = true;
        }
        if (!nuclear && HighLogic.LoadedSceneIsEditor)
        {
            foreach (string fuel in new[] { "EnrichedUranium", "DepletedFuel" })
            {
                if (part.Resources[fuel] == null) continue;
                part.Resources.Remove(fuel);
                resourcesChanged = true;
            }
        }
        // Kerbalism's Unity Start may run after our OnStart; reconcile its
        // pseudo-resource as well, so a passive core cannot produce in background.
        FieldInfo capacity = controller.GetType().GetField("capacity");
        capacity.SetValue(controller, Convert.ChangeType(nuclear ? reactorScale : 0, capacity.FieldType));
        PartResource pseudo = part.Resources["_RWCompactReactor"];
        if (nuclear)
        {
            if (pseudo == null || Math.Abs(pseudo.maxAmount - reactorScale) > 1e-9)
            {
                bool running = (bool)controller.GetType().GetField("running").GetValue(controller);
                Configure(true);
                SetRunning(running);
            }
        }
        else
        {
            if ((bool)controller.GetType().GetField("running").GetValue(controller)) SetRunning(false);
            if (pseudo != null) Configure(false);
        }
        BaseEvent toggle = controller.Events["Toggle"];
        if (toggle != null) toggle.active = nuclear;
        BaseAction action = controller.Actions["Action"];
        if (action != null) action.active = nuclear;
        if (resourcesChanged)
        {
            // Open PAW resource sliders retain old resource references and max
            // ranges. Ship-modified alone does not rebuild those controls.
            GameEvents.onPartResourceListChange.Fire(part);
            if (changed && HighLogic.LoadedSceneIsEditor && EditorLogic.fetch != null && EditorLogic.fetch.ship != null)
                GameEvents.onEditorShipModified.Fire(EditorLogic.fetch.ship);
            MonoUtilities.RefreshContextWindows(part);
            Debug.Log("[RoosterWorks] PWR resource menu rebuilt; actual EC maximum=" + part.Resources["ElectricCharge"].maxAmount);
        }
    }
    private void SetRunning(bool running)
    {
        controller.GetType().GetMethod("SetRunning", new[] { typeof(bool) }).Invoke(controller, new object[] { running });
    }
    private void Configure(bool enabled)
    {
        controller.GetType().GetMethod("Configure", new[] { typeof(bool), typeof(int) }).Invoke(controller, new object[] { enabled, 1 });
    }
    private void Resize(string name, double maximum, double initial)
    {
        PartResource resource = part.Resources[name];
        if (resource == null)
        {
            var node = new ConfigNode("RESOURCE");
            node.AddValue("name", name); node.AddValue("amount", initial); node.AddValue("maxAmount", maximum);
            part.AddResource(node);
            return;
        }
        double fraction = resource.maxAmount > 0 ? resource.amount / resource.maxAmount : 0;
        resource.maxAmount = maximum;
        resource.amount = HighLogic.LoadedSceneIsEditor ? Math.Min(maximum, Math.Max(0, fraction * maximum)) : Math.Min(maximum, resource.amount);
    }
    public override string GetInfo() { return "Battery / Nuclear editor variants. Reactor reserves one of eight battery bays (12.5% less EC storage). Built-in cooling; exterior finish selected independently."; }
}
